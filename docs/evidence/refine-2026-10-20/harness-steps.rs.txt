thread_local! {
    static STEP: Cell<usize> = const { Cell::new(0) };
    static CTRL: RefCell<Option<Retained<Controller>>> = const { RefCell::new(None) };
    static MARK: Cell<usize> = const { Cell::new(0) };
}

fn out() -> String { std::env::var("HW_HARNESS_OUT").unwrap() }

pub fn note(line: String) { if std::env::var("HW_HARNESS_OUT").is_ok() { log(line) } }

fn log(line: String) {
    let mut f = std::fs::OpenOptions::new().create(true).append(true).open(format!("{}/journey.log", out())).unwrap();
    writeln!(f, "[{}] {line}", panel::clock(now_unix())).unwrap();
}

thread_local! { static OPENS: Cell<u32> = const { Cell::new(0) }; }

pub fn start(ctrl: &Retained<Controller>) {
    if std::env::var("HW_OPENS").is_ok() {
        let t = unsafe { NSTimer::timerWithTimeInterval_target_selector_userInfo_repeats(std::env::var("HW_OPENS_START").ok().and_then(|v| v.parse().ok()).unwrap_or(60.0), &**ctrl, sel!(opensStep:), None, false) };
        unsafe { NSRunLoop::mainRunLoop().addTimer_forMode(&t, NSRunLoopCommonModes) };
        return;
    }
    if std::env::var("HW_HARNESS_OUT").is_err() { return; }
    CTRL.with(|c| *c.borrow_mut() = Some(ctrl.clone()));
    later(ctrl, 1.0);
}

fn later(ctrl: &Controller, seconds: f64) {
    let timer = unsafe { NSTimer::timerWithTimeInterval_target_selector_userInfo_repeats(seconds, ctrl, sel!(harnessStep:), None, false) };
    unsafe { NSRunLoop::mainRunLoop().addTimer_forMode(&timer, NSRunLoopCommonModes) };
}

fn state(c: &Controller) -> String {
    let m = c.ivars().monitor.borrow();
    let now = now_unix();
    let latest = m.current().map(|l| format!("{:?} cpu={:?}", l.snapshot.activity.status, l.snapshot.activity.cpu_busy_percent.map(|v| (v * 10.0).round() / 10.0))).unwrap_or("none".into());
    format!("health={:?} routes={:?} paused={} interval={} timer={} history={} latest={} shown={} conclusion={:?}",
        m.health(now), c.ivars().routes.borrow(), m.paused(), m.interval(),
        c.ivars().timer.borrow().as_ref().map_or("none".to_string(), |t| format!("{:.0}s", t.timeInterval())),
        m.history().len(), latest, c.shown(), panel::conclusion(&m, now).1)
}

fn capture(c: &Controller, name: &str) {
    let (Some(root), Some(scroll)) = (c.ivars().root.get(), c.ivars().scroll.get()) else { return };
    let mtm = c.mtm();
    let shot = |view: &NSView, file: &str| {
        let bg = ShapeView::new(mtm, Shape::Fill, view.bounds());
        view.addSubview_positioned_relativeTo(&bg, NSWindowOrderingMode::Below, None);
        let b = view.bounds();
        if let Some(rep) = view.bitmapImageRepForCachingDisplayInRect(b) {
            view.cacheDisplayInRect_toBitmapImageRep(b, &rep);
            let props = NSDictionary::new();
            if let Some(data) = unsafe { rep.representationUsingType_properties(NSBitmapImageFileType::PNG, &props) } {
                std::fs::write(format!("{}/{file}", out()), data.to_vec()).unwrap();
            }
        }
        bg.removeFromSuperview();
    };
    shot(root, &format!("{name}.png"));
    if let Some(doc) = scroll.documentView() {
        if doc.bounds().size.height > scroll.frame().size.height + 1.0 {
            shot(&doc, &format!("{name}-full.png"));
        }
    }
    log(format!("captured {name}"));
}

fn click(c: &Controller, command: &Command) -> bool {
    let Some(root) = c.ivars().root.get() else { return false };
    let index = c.ivars().commands.borrow().iter().position(|x| x == command);
    let Some(view) = index.and_then(|i| root.viewWithTag(i as isize + 1)) else {
        log(format!("click {command:?}: control not found; index={index:?} commands={:?}", c.ivars().commands.borrow()));
        return false;
    };
    let button = view.downcast::<NSButton>().unwrap();
    // Real hit test at the control's center in window coordinates.
    let b = button.bounds();
    let p = button.convertPoint_toView(NSPoint::new(b.size.width / 2.0, b.size.height / 2.0), None);
    let hit = root.window().and_then(|w| w.contentView()).and_then(|cv| unsafe { cv.superview() }.or(Some(cv))).and_then(|frame| frame.hitTest(p));
    let topmost = hit.as_deref().map_or(false, |h| std::ptr::eq(h, &**button as &NSView));
    log(format!("click {command:?} title={:?} enabled={} hit_test_topmost={topmost}", button.title().to_string(), button.isEnabled()));
    unsafe { button.performClick(None) };
    true
}

fn key_escape(c: &Controller) {
    let Some(window) = c.ivars().root.get().and_then(|r| r.window()) else { log("escape: no window".into()); return };
    let event = NSEvent::keyEventWithType_location_modifierFlags_timestamp_windowNumber_context_characters_charactersIgnoringModifiers_isARepeat_keyCode(
        NSEventType::KeyDown, NSPoint::new(0.0, 0.0), NSEventModifierFlags(0), 0.0, window.windowNumber(), None,
        ns_string!("\u{1b}"), ns_string!("\u{1b}"), false, 53).unwrap();
    log(format!("send Escape keyDown; first responder is root: {}", window.firstResponder().map_or(false, |r| std::ptr::eq(&*r as *const _ as *const u8, &**c.ivars().root.get().unwrap() as *const _ as *const u8))));
    window.sendEvent(&event);
}

fn menu(c: &Controller, tag: isize) {
    let item = NSMenuItem::new(c.mtm());
    item.setTag(tag);
    unsafe { let _: () = msg_send![c, menuPerform: &*item]; }
}


impl Controller {
    pub(crate) fn opens_step(&self) {}

    pub(crate) fn harness_step(&self) {
        let step = STEP.with(|s| { let v = s.get(); s.set(v + 1); v });
        let next = |s: f64| later(self, s);
        match step {
            0 => { self.toggle_popover(); log(format!("opened: {}", state(self))); next(6.0) }
            1 => {
                capture(self, "20-home-four-rows");
                let m = self.ivars().monitor.borrow();
                if let Some(l) = m.current() {
                    let d = l.snapshot.disk.as_ref().unwrap();
                    log(format!("disk statfs={} important={:?} thermal={:?}", d.available_bytes, d.important_available_bytes, l.snapshot.thermal));
                }
                drop(m);
                click(self, &Command::Go(Route::Detail(crate::panel::Resource::Thermal)));
                capture(self, "21-temperature");
                click(self, &Command::Back);
                click(self, &Command::Go(Route::Detail(crate::panel::Resource::Disk)));
                capture(self, "22-disk");
                click(self, &Command::Back);
                menu(self, MENU_PROCESSES);
                click(self, &Command::RefreshProcesses);
                next(3.0)
            }
            2 => {
                capture(self, "23-processes-footprint");
                if let crate::monitor::Processes::Done(r) = self.ivars().monitor.borrow().processes() {
                    if let Ok(scan) = &r.outcome {
                        log(format!("processes: footprints MiB = {:?}", scan.top.iter().map(|p| p.footprint_bytes.unwrap_or(0) >> 20).collect::<Vec<_>>()));
                    }
                }
                click(self, &Command::Back);
                // A real analysis of the person's open apps; nothing is quit from it.
                self.handle(Command::Optimize(Group::Memory));
                log(format!("optimize started: routes={:?}", self.ivars().routes.borrow()));
                capture(self, "24-optimize-analyzing");
                next(2.0)
            }
            3 => {
                let summary = match self.ivars().monitor.borrow().optimizer().state() {
                    Some(crate::optimize::Optimization::Ready { candidates, .. }) => format!("ready: {} candidates, selected {}, footprints MiB {:?}", candidates.len(), candidates.iter().filter(|c| c.selected).count(), candidates.iter().map(|c| c.footprint_bytes >> 20).collect::<Vec<_>>()),
                    other => format!("{other:?}"),
                };
                log(format!("memory plan {summary}; plan job joined: {}", self.ivars().plan.borrow().is_none()));
                capture(self, "25-optimize-memory-list");
                self.handle(Command::Optimize(Group::Cpu));
                next(2.5)
            }
            4 => {
                let summary = match self.ivars().monitor.borrow().optimizer().state() {
                    Some(crate::optimize::Optimization::Ready { candidates, .. }) => format!("ready: {:?}", candidates.iter().map(|c| (c.cpu_percent.round(), c.processes)).collect::<Vec<_>>()),
                    other => format!("{other:?}"),
                };
                log(format!("cpu plan {summary}"));
                capture(self, "26-optimize-cpu-list");
                // Quit path on a harmless app the script opened itself, with no documents.
                let ok = std::process::Command::new("open").args(["-g", "-a", "Calculator"]).status().map(|s| s.success()).unwrap_or(false);
                log(format!("launched Calculator: {ok}"));
                next(3.0)
            }
            5 => {
                let pid = std::process::Command::new("pgrep").args(["-x", "Calculator"]).output().ok().and_then(|o| String::from_utf8(o.stdout).ok()).and_then(|s| s.trim().lines().next().and_then(|l| l.parse::<u32>().ok()));
                log(format!("Calculator pid {pid:?}"));
                let Some(pid) = pid else { next(0.5); return };
                {
                    let mut m = self.ivars().monitor.borrow_mut();
                    let o = m.optimizer_mut();
                    o.begin(Group::Memory, now_unix());
                    o.finish(Ok(vec![Candidate { pid, name: "计算器".into(), bundle_id: Some("com.apple.calculator".into()), footprint_bytes: 300 << 20, cpu_percent: 0.0, processes: 1, active: false, selected: true }]), now_unix());
                }
                if !self.shown() {
                    self.toggle_popover();
                }
                self.navigate(Route::Optimize(Group::Memory));
                self.render(false);
                capture(self, "27-optimize-calculator-selected");
                click(self, &Command::QuitSelected);
                log(format!("after quit click: {:?}", self.ivars().monitor.borrow().optimizer().state()));
                capture(self, "28-optimize-requested");
                next(22.0)
            }
            6 => {
                log(format!("after verify: {:?}", self.ivars().monitor.borrow().optimizer().state()));
                let still = std::process::Command::new("pgrep").args(["-x", "Calculator"]).status().map(|s| s.success()).unwrap_or(false);
                log(format!("Calculator still running per pgrep: {still}"));
                capture(self, "29-optimize-verified");
                next(0.5)
            }
            _ => { log("quit via More menu item".into()); menu(self, MENU_QUIT); }
        }
    }
}
